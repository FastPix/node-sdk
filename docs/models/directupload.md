# DirectUpload

Displays the result of the request.

## Example Usage

```typescript
import { DirectUpload } from "@fastpix/fastpix-node/models";

let value: DirectUpload = {
  uploadId: "your-upload-id",
  trial: false,
  status: "waiting",
  url:
    "{\"url\":\"your-upload-url\"}",
  corsOrigin: "*",
  pushMediaSettings: {
    playbackIds: [
      {
        id: "your-playback-id",
      },
    ],
    metadata: {
      "key1": "value1",
    },
    mediaQuality: "standard",
    sourceAccess: false,
    optimizeAudio: false,
  },
};
```
