let value: SourceAccessMedia = {
  thumbnail:
    "your-thumbnail-url",
  id: "your-id",

    "key1": "value1",

      id: "your-id",

  tracks: [
    {
      id: "your-id",
      type: "audio",
      status: "available",
      languageName: "your-language-name",
      languageCode: "your-language-code",
    },
  ],
  generatedSubtitles: [
    {
      status: "preparing",

    status: "ready",

    status: "ready",
  },

};
